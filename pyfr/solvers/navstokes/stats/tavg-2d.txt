# Navier-Stokes statistics, 2D
# Budget terms follow Pope (2000), sec. 7.3.5; all viscous
# contractions are accumulated mu-weighted so that constant
# and Sutherland viscosity share one set of definitions

# Mean viscosity
field mu = mu
# Solver-consistent mean velocity and pressure gradients
field gux = grad_u_x
field guy = grad_u_y
field gvx = grad_v_x
field gvy = grad_v_y
field gpx = grad_p_x
field gpy = grad_p_y
# Gradients of the velocity second moments
field duux = 2*u*grad_u_x
field duuy = 2*u*grad_u_y
field duvx = grad_u_x*v + u*grad_v_x
field duvy = grad_u_y*v + u*grad_v_y
field dvvx = 2*v*grad_v_x
field dvvy = 2*v*grad_v_y
# Mu-weighted velocity gradient contractions
field euu = mu*(grad_u_x*grad_u_x + grad_u_y*grad_u_y)
field euv = mu*(grad_u_x*grad_v_x + grad_u_y*grad_v_y)
field evv = mu*(grad_v_x*grad_v_x + grad_v_y*grad_v_y)

# Transposed contraction for the true dissipation
field eet
    mu*(grad_u_x*grad_u_x + 2*grad_u_y*grad_v_x + grad_v_y*grad_v_y)

# Pressure-gradient and pressure-dilatation moments
field pdux = p*grad_u_x
field dpux = grad_p_x*u + p*grad_u_x
field pduy = p*grad_u_y
field dpuy = grad_p_y*u + p*grad_u_y
field pdvx = p*grad_v_x
field dpvx = grad_p_x*v + p*grad_v_x
field pdvy = p*grad_v_y
field dpvy = grad_p_y*v + p*grad_v_y
# Divergence of the triple velocity products

field dtuu = u*u*(3*grad_u_x + grad_v_y) + 2*u*v*grad_u_y

field dtuv = 2*u*v*(grad_u_x + grad_v_y) + u*u*grad_v_x + v*v*grad_u_y

field dtvv = v*v*(grad_u_x + 3*grad_v_y) + 2*u*v*grad_v_x

# Compressibility moments

# Mean dilatation
field d
    (grad_u_x + grad_v_y)

field dd = (grad_u_x + grad_v_y)*(grad_u_x + grad_v_y)

# Enstrophy
field enst
    (grad_v_x - grad_u_y)*(grad_v_x - grad_u_y)

field pdil = p*(grad_u_x + grad_v_y)
field pr = p*rho
field rr = rho*rho
# Centred Reynolds stresses; unclamped for budget use
derived-hidden rcuu = uu - u*u
derived-hidden rcuv = uv - u*v
derived-hidden rcvv = vv - v*v
# Reynolds stress transport budgets

# P_ij = -⟨ui'uk'⟩ ∂Uj/∂xk - ⟨uj'uk'⟩ ∂Ui/∂xk
derived rs-uu-production
    -2*(rcuu*gux + rcuv*guy)

# C_ij = Uk ∂⟨ui'uj'⟩/∂xk
derived rs-uu-convection
    u*(duux - 2*u*gux) + v*(duuy - 2*u*guy)

# T_ij = -∂⟨ui'uj'uk'⟩/∂xk via runtime flux divergences
derived rs-uu-turb-transport
    -(dtuu - (3*(gux*uu + u*duux) - 6*u*u*gux + 2*(guy*uv + u*duvy) +
    gvy*uu + v*duuy - 2*(2*u*v*guy + u*u*gvy)))

# PT_ij = -(∂⟨p'uj'⟩/∂xi + ∂⟨p'ui'⟩/∂xj)/ρ
derived rs-uu-press-transport
    -2*(dpux - (gpx*u + p*gux))/rho

# R_ij = ⟨p'(∂ui'/∂xj + ∂uj'/∂xi)⟩/ρ
derived rs-uu-press-strain
    2*(pdux - p*gux)/rho

# D_ij = ν ∇²⟨ui'uj'⟩
derived rs-uu-visc-diffusion
    (mu/rho)*lap_rcuu

# ε_ij = 2ν ⟨∂ui'/∂xk ∂uj'/∂xk⟩
derived rs-uu-dissipation
    2*(euu - mu*(gux*gux + guy*guy))/rho

# Closure defect; should vanish when converged
derived rs-uu-residual
    rs-uu-production + rs-uu-press-strain + rs-uu-press-transport +
    rs-uu-turb-transport + rs-uu-visc-diffusion - rs-uu-dissipation -
    rs-uu-convection

set budget-uu
    rs-uu-production rs-uu-convection rs-uu-turb-transport
    rs-uu-press-transport rs-uu-press-strain rs-uu-visc-diffusion
    rs-uu-dissipation rs-uu-residual

# P_ij = -⟨ui'uk'⟩ ∂Uj/∂xk - ⟨uj'uk'⟩ ∂Ui/∂xk
derived rs-uv-production
    -(rcuu*gvx + rcuv*(gux + gvy) + rcvv*guy)

# C_ij = Uk ∂⟨ui'uj'⟩/∂xk
derived rs-uv-convection
    (u*(duvx - v*gux - u*gvx) + v*(duvy - v*guy - u*gvy))

# T_ij = -∂⟨ui'uj'uk'⟩/∂xk via runtime flux divergences
derived rs-uv-turb-transport
    -(dtuv - (2*(gux*uv + u*duvx) + gvx*uu + v*duux - 2*(2*u*v*gux +
    u*u*gvx) + guy*vv + u*dvvy + 2*(gvy*uv + v*duvy) - 2*(v*v*guy +
    2*u*v*gvy)))

# PT_ij = -(∂⟨p'uj'⟩/∂xi + ∂⟨p'ui'⟩/∂xj)/ρ
derived rs-uv-press-transport
    -(dpuy - (gpy*u + p*guy) + dpvx - (gpx*v + p*gvx))/rho

# R_ij = ⟨p'(∂ui'/∂xj + ∂uj'/∂xi)⟩/ρ
derived rs-uv-press-strain
    (pduy + pdvx - p*(guy + gvx))/rho

# D_ij = ν ∇²⟨ui'uj'⟩
derived rs-uv-visc-diffusion
    (mu/rho)*lap_rcuv

# ε_ij = 2ν ⟨∂ui'/∂xk ∂uj'/∂xk⟩
derived rs-uv-dissipation
    2*(euv - mu*(gux*gvx + guy*gvy))/rho

# Closure defect; should vanish when converged
derived rs-uv-residual
    rs-uv-production + rs-uv-press-strain + rs-uv-press-transport +
    rs-uv-turb-transport + rs-uv-visc-diffusion - rs-uv-dissipation -
    rs-uv-convection

set budget-uv
    rs-uv-production rs-uv-convection rs-uv-turb-transport
    rs-uv-press-transport rs-uv-press-strain rs-uv-visc-diffusion
    rs-uv-dissipation rs-uv-residual

# P_ij = -⟨ui'uk'⟩ ∂Uj/∂xk - ⟨uj'uk'⟩ ∂Ui/∂xk
derived rs-vv-production
    -2*(rcuv*gvx + rcvv*gvy)

# C_ij = Uk ∂⟨ui'uj'⟩/∂xk
derived rs-vv-convection
    u*(dvvx - 2*v*gvx) + v*(dvvy - 2*v*gvy)

# T_ij = -∂⟨ui'uj'uk'⟩/∂xk via runtime flux divergences
derived rs-vv-turb-transport
    -(dtvv - (2*(gvx*uv + v*duvx) + gux*vv + u*dvvx - 2*(2*u*v*gvx +
    v*v*gux) + 3*(gvy*vv + v*dvvy) - 6*v*v*gvy))

# PT_ij = -(∂⟨p'uj'⟩/∂xi + ∂⟨p'ui'⟩/∂xj)/ρ
derived rs-vv-press-transport
    -2*(dpvy - (gpy*v + p*gvy))/rho

# R_ij = ⟨p'(∂ui'/∂xj + ∂uj'/∂xi)⟩/ρ
derived rs-vv-press-strain
    2*(pdvy - p*gvy)/rho

# D_ij = ν ∇²⟨ui'uj'⟩
derived rs-vv-visc-diffusion
    (mu/rho)*lap_rcvv

# ε_ij = 2ν ⟨∂ui'/∂xk ∂uj'/∂xk⟩
derived rs-vv-dissipation
    2*(evv - mu*(gvx*gvx + gvy*gvy))/rho

# Closure defect; should vanish when converged
derived rs-vv-residual
    rs-vv-production + rs-vv-press-strain + rs-vv-press-transport +
    rs-vv-turb-transport + rs-vv-visc-diffusion - rs-vv-dissipation -
    rs-vv-convection

set budget-vv
    rs-vv-production rs-vv-convection rs-vv-turb-transport
    rs-vv-press-transport rs-vv-press-strain rs-vv-visc-diffusion
    rs-vv-dissipation rs-vv-residual

# TKE budget terms as half-traces of the stress budgets

# P_ij = -⟨ui'uk'⟩ ∂Uj/∂xk - ⟨uj'uk'⟩ ∂Ui/∂xk
derived tke-production
    -(rcuu*gux + rcuv*(guy + gvx) + rcvv*gvy)

# C_ij = Uk ∂⟨ui'uj'⟩/∂xk
derived tke-convection
    0.5*(u*(duux - 2*u*gux + dvvx - 2*v*gvx) + v*(duuy - 2*u*guy +
    dvvy - 2*v*gvy))

# T_ij = -∂⟨ui'uj'uk'⟩/∂xk via runtime flux divergences
derived tke-turb-transport
    0.5*(-(dtuu - (3*(gux*uu + u*duux) - 6*u*u*gux + 2*(guy*uv +
    u*duvy) + gvy*uu + v*duuy - 2*(2*u*v*guy + u*u*gvy))) - (dtvv -
    (2*(gvx*uv + v*duvx) + gux*vv + u*dvvx - 2*(2*u*v*gvx + v*v*gux) +
    3*(gvy*vv + v*dvvy) - 6*v*v*gvy)))

# PT_ij = -(∂⟨p'uj'⟩/∂xi + ∂⟨p'ui'⟩/∂xj)/ρ
derived tke-press-transport
    -(dpux - (gpx*u + p*gux) + dpvy - (gpy*v + p*gvy))/rho

# R_ij = ⟨p'(∂ui'/∂xj + ∂uj'/∂xi)⟩/ρ
derived tke-press-dilatation
    (pdux + pdvy - p*(gux + gvy))/rho

# D_ij = ν ∇²⟨ui'uj'⟩
derived tke-visc-diffusion
    0.5*(mu/rho)*(lap_rcuu + lap_rcvv)

# ε_ij = 2ν ⟨∂ui'/∂xk ∂uj'/∂xk⟩
derived eps
    (euu + evv - mu*(gux*gux + guy*guy + gvx*gvx + gvy*gvy))/rho

# Closure defect; should vanish when converged
derived tke-residual
    tke-production + tke-press-dilatation + tke-press-transport +
    tke-turb-transport + tke-visc-diffusion - eps - tke-convection

set budget-k
    tke-production tke-convection tke-turb-transport tke-press-transport
    tke-press-dilatation tke-visc-diffusion eps tke-residual

# Turbulence length, time and velocity scales

# Kolmogorov length scale
derived eta
    pow(pow(mu/rho, 3)/eps, 0.25)

# Kolmogorov time scale
derived tau-eta
    sqrt((mu/rho)/eps)

# Kolmogorov velocity scale
derived u-eta
    pow((mu/rho)*eps, 0.25)

# Taylor microscale
derived taylor
    sqrt(10*(mu/rho)*tke/eps)

# Taylor-microscale Reynolds number
derived re-lambda
    sqrt(2*tke/3)*taylor/(mu/rho)

# Integral length scale estimate
derived l-int
    pow(tke, 1.5)/eps

# Grid spacing per solution point over eta
derived res-eta
    grid_hmax/(porder + 1)/eta

derived res-eta-min = grid_hmin/(porder + 1)/eta
derived res-taylor = grid_hmax/(porder + 1)/taylor

set scales
    eps eta tau-eta u-eta taylor re-lambda l-int res-eta res-eta-min
    res-taylor

# Dissipation upgraded with the transposed term
derived eps-true
    eps + (eet - mu*(gux*gux + 2*guy*gvx + gvy*gvy))/rho

set scales-exact
    scales eps-true

# Velocity-gradient tensor invariants

# Second invariant Q = det(A) in 2D
field invq
    grad_u_x*grad_v_y - grad_u_y*grad_v_x

field invqq
    (grad_u_x*grad_v_y - grad_u_y*grad_v_x)*(grad_u_x*grad_v_y -
    grad_u_y*grad_v_x)

# Q of the mean velocity gradient tensor
derived q-mean
    gux*gvy - guy*gvx

derived q-rms = sqrt(max(invqq - invq*invq, 0))

set vgt
    invq q-rms q-mean

# Wall statistics from mean mu-weighted velocity gradients
field mgux = mu*grad_u_x
field mguy = mu*grad_u_y
field mgvx = mu*grad_v_x
field mgvy = mu*grad_v_y
field mqx = (mu/Pr)*grad_cpT_x
field mqy = (mu/Pr)*grad_cpT_y
# Mean wall quantities; boundary exports only

# Mean viscous traction component
derived trac-x
    2*mgux*norm_x + (mguy + mgvx)*norm_y

# Mean viscous traction component
derived trac-y
    (mgvx + mguy)*norm_x + 2*mgvy*norm_y

# Normal component of the mean traction
derived trac-n
    trac-x*norm_x + trac-y*norm_y

# Magnitude of the mean tangential traction
derived tau-wall
    sqrt((trac-x - trac-n*norm_x)*(trac-x - trac-n*norm_x) + (trac-y -
    trac-n*norm_y)*(trac-y - trac-n*norm_y))

# Mean skin friction coefficient
derived cf requires rho-inf u-inf
    tau-wall/(0.5*rho-inf*u-inf*u-inf)

# Friction velocity from the mean traction
derived u-tau
    sqrt(tau-wall/rho)

# y+ of the first interior solution point
derived yplus
    boundary_dist*u-tau*rho/mu

# Mean wall heat flux; positive into the wall
derived q-wall
    -(mqx*norm_x + mqy*norm_y)

set wall
    trac-x trac-y trac-n tau-wall cf u-tau yplus

set wall-heat
    wall q-wall

# Compressibility diagnostics

# Solenoidal dissipation from fluctuating vorticity
derived eps-sol
    (mu/rho)*(enst - ((gvx - guy)*(gvx - guy)))

# Dilatational dissipation
derived eps-dil
    (4/3)*(mu/rho)*(dd - d*d)

# Pressure-dilatation correlation
derived press-dilatation
    (pdil - p*(gux + gvy))/rho

# Turbulent Mach number
derived mach-t
    sqrt(2*tke*rho/(gamma*p))

derived rhorms = sqrt(max(rr - rho*rho, 0))

# Isentropic pressure-density correlation
derived isen-corr
    (pr - p*rho)/(prms*rhorms)

set compressibility
    eps-sol eps-dil press-dilatation mach-t rhorms isen-corr
